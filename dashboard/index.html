int oledPage = 0;
unsigned long lastOledSwitch = 0;

bool emergencyStop = false;

void setup() {
  Serial.begin(9600);
  Wire.begin();

  // OLED Init
  if (!oled.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    Serial.println(F("{\"err\":\"OLED_FAIL\"}"));
  }
  oled.clearDisplay();
  oled.setTextColor(SSD1306_WHITE);
  oled.setTextSize(1);
  oled.setCursor(10, 20);
  oled.println(F("ARIES ROVER v2.0"));
  oled.setCursor(20, 36);
  oled.println(F("Initializing..."));
  oled.display();

  // Sensor Init
  dht.begin();
  ds18b20.begin();

  if (!bmp.begin(0x76)) {
    Serial.println(F("{\"err\":\"BMP280_FAIL\"}"));
  }
  bmp.setSampling(Adafruit_BMP280::MODE_NORMAL,
                  Adafruit_BMP280::SAMPLING_X2,
                  Adafruit_BMP280::SAMPLING_X16,
                  Adafruit_BMP280::FILTER_X16,
                  Adafruit_BMP280::STANDBY_MS_500);

  if (!particleSensor.begin(Wire, I2C_SPEED_FAST)) {
    Serial.println(F("{\"err\":\"MAX30102_FAIL\"}"));
  } else {
    particleSensor.setup();
    particleSensor.setPulseAmplitudeRed(0x0A);
    particleSensor.setPulseAmplitudeGreen(0);
  }

  // Motor Shield Pins (L293D)
  // Motor shield manages its own pins via library or manual D3,D4,D5,D6,D7,D8,D11,D12
  pinMode(TRIG1, OUTPUT); pinMode(ECHO1, INPUT);

  Serial.println(F("{\"status\":\"ARIES_ONLINE\",\"sensors\":\"ALL_OK\"}"));
}

long getDist() {
  digitalWrite(TRIG1, LOW); delayMicroseconds(2);
  digitalWrite(TRIG1, HIGH); delayMicroseconds(10);
  digitalWrite(TRIG1, LOW);
  long d = pulseIn(ECHO1, HIGH, 23500);
  return (d == 0) ? 999 : d * 0.034 / 2;
}

void stopMotors() {
  // L293D Shield: set all motor outputs LOW
  for (int p = 4; p <= 8; p++) { pinMode(p, OUTPUT); digitalWrite(p, LOW); }
  emergencyStop = true;
}

void updateHeartRate() {
  long irVal = particleSensor.getIR();
  if (checkForBeat(irVal)) {
    long delta = millis() - lastBeat;
    lastBeat = millis();
    beatsPerMinute = 60 / (delta / 1000.0);
    if (beatsPerMinute > 20 && beatsPerMinute < 255) {
      rates[rateSpot++] = (byte)beatsPerMinute;
      rateSpot %= 4;
      beatAvg = 0;
      for (byte x = 0; x < 4; x++) beatAvg += rates[x];
      beatAvg /= 4;
    }
  }
}

void updateOLED(float t, float t2, int gas, long dist, int hr, float pres) {
  if (millis() - lastOledSwitch > 3000) {
    oledPage = (oledPage + 1) % 3;
    lastOledSwitch = millis();
  }
  oled.clearDisplay();
  oled.setCursor(0, 0);
  oled.setTextSize(1);

  if (oledPage == 0) {
    oled.println(F("== ARIES ROVER v2 =="));
    oled.print(F("Gas: ")); oled.print(gas); oled.println(F(" PPM"));
    oled.print(F("T1:")); oled.print(t,1); oled.print(F("C T2:")); oled.print(t2,1); oled.println(F("C"));
    oled.print(F("Dist: ")); oled.print(dist); oled.println(F(" cm"));
    if (dist < 25) { oled.setTextSize(1); oled.println(F("!! OBSTACLE STOP !!")); }
  } else if (oledPage == 1) {
    oled.setTextSize(1); oled.println(F("=== VITALS ==="));
    oled.setTextSize(2);
    oled.print(F("HR:")); oled.println(hr);
    oled.setTextSize(1);
    oled.println(F("BPM (MAX30102)"));
    oled.print(F("Pres:")); oled.print(pres,1); oled.println(F("hPa"));
  } else {
    int roll = 0; // Read from MPU if needed
    oled.println(F("=== IMU STATUS ==="));
    if (emergencyStop) {
      oled.setTextSize(1);
      oled.println(F("!! TILT ALERT !!"));
      oled.println(F("AUTO-STOP ACTIVE"));
    } else {
      oled.println(F("TERRAIN: STABLE"));
    }
  }
  oled.display();
}

void streamTelemetry() {
  float t = dht.readTemperature();
  float h = dht.readHumidity();
  ds18b20.requestTemperatures();
  float t2 = ds18b20.getTempCByIndex(0);
  int gas = analogRead(MQ135_PIN);
  long dist = getDist();
  float pres = bmp.readPressure() / 100.0F;
  float alt  = bmp.readAltitude(1013.25);

  // Safety auto-stop
  if (dist < 25 || gas > 600) stopMotors();

  updateHeartRate();
  updateOLED(t, t2, gas, dist, beatAvg, pres);

  Serial.print(F("{\"t\":")); Serial.print(t,1);
  Serial.print(F(",\"t2\":")); Serial.print(t2,1);
  Serial.print(F(",\"h\":")); Serial.print(h,1);
  Serial.print(F(",\"gas\":")); Serial.print(gas);
  Serial.print(F(",\"dist\":")); Serial.print(dist);
  Serial.print(F(",\"pres\":")); Serial.print(pres,1);
  Serial.print(F(",\"alt\":")); Serial.print(alt,1);
  Serial.print(F(",\"hr\":")); Serial.print(beatAvg);
  Serial.println(F("}"));
}

void loop() {
  updateHeartRate();  // Poll MAX30102 continuously

  if (Serial.available()) {
    char c = Serial.read();
    emergencyStop = false;
    if      (c=='F') { /* drive forward  */ }
    else if (c=='B') { /* drive backward */ }
    else if (c=='L') { /* turn left      */ }
    else if (c=='R') { /* turn right     */ }
    else if (c=='S') stopMotors();
    else if (c=='D') streamTelemetry();
  }

  if (millis() - lastTelemetry > 500) {
    streamTelemetry();
    lastTelemetry = millis();
  }
}
